+ export interface RetryOptions {
+   attempts?: number;
+   baseMs?: number;
+   maxMs?: number;
+ }
+ 
+ const RETRYABLE = new Set(["SlowDown", "ServiceUnavailable", "InternalError", "RequestTimeout"]);
+ 
+ export async function withRetry<T>(fn: () => Promise<T>, opts: RetryOptions = {}): Promise<T> {
+   const { attempts = 5, baseMs = 100, maxMs = 3200 } = opts;
+   for (let i = 1; ; i++) {
+     try {
+       return await fn();
+     } catch (err) {
+       const code = (err as { name?: string }).name ?? "";
+       if (i >= attempts || !RETRYABLE.has(code)) throw err;
+       const delay = Math.min(maxMs, baseMs * 2 ** (i - 1));
+       await sleep(delay / 2 + Math.random() * (delay / 2));
+     }
+   }
+ }
+ 
+ const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
